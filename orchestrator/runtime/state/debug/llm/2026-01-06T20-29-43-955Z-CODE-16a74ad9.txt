### Krok 3: Vytvoření WebSocket serveru

Vytvoříme soubor `server.js` v adresáři `/tmp/dual-test/`. Tento soubor bude obsahovat kód pro WebSocket server.

```javascript
// /tmp/dual-test/server.js
const express = require('express');
const http = require('http');
const WebSocket = require('ws');

const app = express();
const server = http.createServer(app);
const wss = new WebSocket.Server({ server });

let clients = [];
let messagesHistory = [];

wss.on('connection', (ws) => {
    let nickname;

    ws.on('message', (message) => {
        const data = JSON.parse(message);

        if (data.type === 'nickname') {
            nickname = data.nickname;
            clients.push({ ws, nickname });
            broadcast({ type: 'join', nickname });
            updateClientsList();
            sendHistory(ws);
        } else if (data.type === 'message') {
            messagesHistory.push({ nickname, message: data.message });
            if (messagesHistory.length > 50) {
                messagesHistory.shift();
            }
            broadcast({ type: 'message', nickname, message: data.message });
        }
    });

    ws.on('close', () => {
        clients = clients.filter(client => client.ws !== ws);
        broadcast({ type: 'leave', nickname });
        updateClientsList();
    });
});

function broadcast(data) {
    wss.clients.forEach((client) => {
        if (client.readyState === WebSocket.OPEN) {
            client.send(JSON.stringify(data));
        }
    });
}

function updateClientsList() {
    const clientsList = clients.map(client => client.nickname);
    broadcast({ type: 'clients', clients: clientsList });
}

function sendHistory(ws) {
    ws.send(JSON.stringify({ type: 'history', messages: messagesHistory }));
}

server.listen(3000, () => {
    console.log('Server is listening on port 3000');
});
```

### Krok 5: Vytvoření WebSocket klienta

Vytvoříme soubor `client.js` v adresáři `/tmp/dual-test/public/`. Tento soubor bude obsahovat kód pro WebSocket klienta.

```javascript
// /tmp/dual-test/public/client.js
const ws = new WebSocket('ws://localhost:3000');
let nickname;

function setNickname() {
    const input = document.getElementById('nickname');
    nickname = input.value;
    ws.send(JSON.stringify({ type: 'nickname', nickname }));
    input.disabled = true;
}

function sendMessage() {
    const input = document.getElementById('input');
    const message = input.value;
    if (message) {
        ws.send(JSON.stringify({ type: 'message', message }));
        input.value = '';
    }
}

ws.onmessage = function(event) {
    const data = JSON.parse(event.data);
    const messagesDiv = document.getElementById('messages');

    if (data.type === 'join') {
        messagesDiv.innerHTML += `<div><strong>${data.nickname}</strong> joined the chat.</div>`;
    } else if (data.type === 'leave') {
        messagesDiv.innerHTML += `<div><strong>${data.nickname}</strong> left the chat.</div>`;
    } else if (data.type === 'message') {
        messagesDiv.innerHTML += `<div><strong>${data.nickname}:</strong> ${data.message}</div>`;
    } else if (data.type === 'clients') {
        const clientsDiv = document.getElementById('clients');
        clientsDiv.innerHTML = '<h3>Online Users:</h3>' + data.clients.map(client => `<div>${client}</div>`).join('');
    } else if (data.type === 'history') {
        data.messages.forEach(msg => {
            messagesDiv.innerHTML += `<div><strong>${msg.nickname}:</strong> ${msg.message}</div>`;
        });
    }

    messagesDiv.scrollTop = messagesDiv.scrollHeight;
};
```
