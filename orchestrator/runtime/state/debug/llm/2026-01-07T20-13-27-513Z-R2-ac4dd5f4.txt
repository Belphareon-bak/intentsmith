```javascript
// /path/to/project/todo.js
const fs = require('fs');
const path = '/tmp/workflow-auto/tasks.json';

function loadTasks() {
  try {
    const data = fs.readFileSync(path, 'utf8');
    return JSON.parse(data);
  } catch (err) {
    return [];
  }
}

function saveTasks(tasks) {
  fs.writeFileSync(path, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
  const tasks = loadTasks();
  const newTask = {
    id: tasks.length + 1,
    text: text,
    done: false,
    createdAt: new Date().toISOString()
  };
  tasks.push(newTask);
  saveTasks(tasks);
  console.log(`Úkol "${text}" přidán.`);
}

function listTasks() {
  const tasks = loadTasks();
  tasks.forEach(task => {
    console.log(`[${task.done ? 'x' : ' '}] ${task.id}. ${task.text}`);
  });
}

function markTaskDone(id) {
  const tasks = loadTasks();
  const taskIndex = tasks.findIndex(task => task.id === parseInt(id));
  if (taskIndex !== -1) {
    tasks[taskIndex].done = true;
    saveTasks(tasks);
    console.log(`Úkol ${id} označen jako hotový.`);
  } else {
    console.log('Úkol s tímto ID neexistuje.');
  }
}

if (process.argv.length < 3) {
  console.log('Zadejte příkaz: add, list nebo done');
  process.exit(1);
}

const command = process.argv[2];
switch(command) {
  case 'add':
    if (process.argv.length < 4) {
      console.log('Prosím zadejte text úkolu.');
      break;
    }
    addTask(process.argv.slice(3).join(' '));
    break;
  case 'list':
    listTasks();
    break;
  case 'done':
    if (process.argv.length < 4) {
      console.log('Prosím zadejte ID úkolu.');
      break;
    }
    markTaskDone(process.argv[3]);
    break;
  default:
    console.log('Neznámý příkaz');
}
```
