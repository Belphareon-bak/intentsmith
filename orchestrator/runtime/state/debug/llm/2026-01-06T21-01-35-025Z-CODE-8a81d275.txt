        document.addEventListener('DOMContentLoaded', () => {
            const sidebar = document.getElementById('sidebar');
            const toggleSidebarBtn = document.getElementById('toggle-sidebar-btn');
            const sendBtn = document.getElementById('send-btn');
            const userInput = document.getElementById('user-input');
            const chatMessages = document.getElementById('chat-messages');
            const loadingSpinner = document.getElementById('loading-spinner');

            toggleSidebarBtn.addEventListener('click', () => {
                sidebar.classList.toggle('visible');

            sendBtn.addEventListener('click', sendMessage);
            userInput.addEventListener('keypress', (e) => {
                if (e.key === 'Enter') {
                    sendMessage();
                }

            function sendMessage() {
                const message = userInput.value.trim();
                if (!message) return;

                userInput.value = '';
                loadingSpinner.style.display = 'inline-block';
                sendBtn.disabled = true;

                const sessionId = 'session1'; // Replace with actual session ID
                const workdir = '/path/to/workdir'; // Replace with actual work directory

                fetch('/chat', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({ message, sessionId, workdir })
                })
                .then(response => response.json())
                .then(data => {
                    renderMessage(message, 'user');
                    renderMessage(data.response, 'bot');
                })
                .catch(error => {
                    console.error('Error:', error);
                    renderMessage('An error occurred. Please try again.', 'bot');
                })
                .finally(() => {
                    loadingSpinner.style.display = 'none';
                    sendBtn.disabled = false;
                });
            }

            function renderMessage(message, type) {
                const messageElement = document.createElement('div');
                messageElement.classList.add('message', type);

                const parsedMessage = marked.parse(message);
                messageElement.innerHTML = parsedMessage;

                const codeBlocks = messageElement.querySelectorAll('pre code');
                codeBlocks.forEach(block => {
                    hljs.highlightBlock(block);

                    const copyButton = document.createElement('button');
                    copyButton.textContent = 'Copy';
                    copyButton.addEventListener('click', () => {
                        navigator.clipboard.writeText(block.innerText);
                    });
                    block.parentElement.appendChild(copyButton);
                });

                chatMessages.appendChild(messageElement);
                chatMessages.scrollTop = chatMessages.scrollHeight;
            }
        });
    </script>
