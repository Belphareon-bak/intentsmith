```javascript
const fs = require('fs');
const path = require('path');

// Cesta k souboru tasks.json
const TASKS_FILE = path.join('/tmp', 'workflow-auto', 'tasks.json');

function loadTasks() {
    try {
        const data = fs.readFileSync(TASKS_FILE, 'utf8');
        return JSON.parse(data);
    } catch (error) {
        if (error.code === 'ENOENT') {
            // Soubor neexistuje, vytvoříme prázdný pole
            return [];
        }
        throw error;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(TASKS_FILE, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const lastId = tasks.length > 0 ? Math.max(...tasks.map(task => task.id)) : 0;
    const newTask = {
        id: lastId + 1,
        text: text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
    console.log('Úkol přidán.');
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly nenajdено.');
        return;
    }
    console.log('\nSeznam úkolů:');
    tasks.forEach((task, index) => {
        const doneSymbol = task.done ? '[x]' : '[ ]';
        console.log(`\n${index + 1}. ${doneSymbol} ID: ${task.id}`);
        console.log(`   Text: ${task.text}`);
        console.log(`   Vytvořeno: ${new Date(task.createdAt).toLocaleString()}`);
    });
}

function doneTask(id) {
    const tasks = loadTasks();
    const taskIndex = tasks.findIndex(task => task.id === id);
    if (taskIndex === -1) {
        console.log('Úkol s daným ID nebyl nalezen.');
        return;
    }
    tasks[taskIndex].done = true;
    saveTasks(tasks);
    console.log('Úkol označen jako hotový.');
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.log('Zadejte příkaz: add <text>, list, nebo done <id>');
        return;
    }

    const command = args[0];
    switch (command.toLowerCase()) {
        case 'add':
            if (args.length < 2) {
                console.log('Příkaz "add" vyžaduje text úkolu.');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;

        case 'list':
            listTasks();
            break;

        case 'done':
            if (args.length < 2) {
                console.log('Příkaz "done" vyžaduje ID úkolu.');
                return;
            }
            const taskId = parseInt(args[1], 10);
            if (isNaN(taskId)) {
                console.log('ID musí být číslo.');
                return;
            }
            doneTask(taskId);
            break;

        default:
            console.log('Neznámý příkaz. Možné příkazy: add, list, done.');
    }
}

main();
```
