#### Vytvoření souboru `/tmp/test/todo.js`

```javascript
const fs = require('fs');
const path = '/tmp/test/todo.json';

// Zajištění existenci adresáře a souboru
if (!fs.existsSync('/tmp/test/')) {
    fs.mkdirSync('/tmp/test/');
}

if (!fs.existsSync(path)) {
    fs.writeFileSync(path, JSON.stringify([]));
}

function loadTodos() {
    return JSON.parse(fs.readFileSync(path, 'utf8'));
}

function saveTodos(todos) {
    fs.writeFileSync(path, JSON.stringify(todos, null, 2));
}

function addTodo(task) {
    const todos = loadTodos();
    todos.push({ task, done: false });
    saveTodos(todos);
    console.log(`Úkol "${task}" byl přidán.`);
}

function listTodos() {
    const todos = loadTodos();
    todos.forEach((todo, index) => {
        console.log(`${index + 1}. [${todo.done ? 'x' : ' '}] ${todo.task}`);
    });
}

function doneTodo(index) {
    const todos = loadTodos();
    if (todos[index - 1]) {
        todos[index - 1].done = true;
        saveTodos(todos);
        console.log(`Úkol číslo ${index} byl označen jako dokončený.`);
    } else {
        console.log('Neplatné číslo úkolu.');
    }
}

const command = process.argv[2];
const argument = process.argv[3];

switch (command) {
    case 'add':
        if (argument) {
            addTodo(argument);
        } else {
            console.log('Prosím, zadejte název úkolu.');
        }
        break;
    case 'list':
        listTodos();
        break;
    case 'done':
        if (argument && !isNaN(argument)) {
            doneTodo(parseInt(argument, 10));
        } else {
            console.log('Prosím, zadejte platné číslo úkolu.');
        }
        break;
    default:
        console.log('Neznámý příkaz. Použijte "add", "list" nebo "done".');
}
```
