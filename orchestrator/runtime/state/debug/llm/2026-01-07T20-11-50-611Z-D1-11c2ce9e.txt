```javascript
const fs = require('fs');
const path = require('path');

const tasksPath = path.join('/tmp', 'workflow-auto', 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (err) {
        if (err.code === 'ENOENT') {
            return [];
        }
        throw err;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: Date.now().toString(),
        text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
    console.log('Úkol byl přidán.');
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly nenajdено.');
        return;
    }
    tasks.forEach((task, index) => {
        const status = task.done ? '[x]' : '[ ]';
        console.log(`${index + 1}. ${status} ${task.text}`);
    });
}

function markDone(id) {
    const tasks = loadTasks();
    const foundTask = tasks.find(task => task.id === id);
    if (!foundTask) {
        console.log('Úkol s daným ID nebyl nalezen.');
        return;
    }
    foundTask.done = true;
    saveTasks(tasks);
    console.log('Úkol byl označen jako hotový.');
}

function showHelp() {
    console.log('Usage: node todo.js [add|list|done]');
    console.log('Commands:');
    console.log('- add <text> - přidá úkol');
    console.log('- list - vypíše úkoly');
    console.log('- done <id> - označí úkol jako hotový');
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        showHelp();
        return;
    }

    const command = args[0];
    switch (command.toLowerCase()) {
        case 'add':
            if (args.length < 2) {
                console.log('Chybí text úkolu.');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;
        case 'list':
            listTasks();
            break;
        case 'done':
            if (args.length < 2) {
                console.log('Chybí ID úkolu.');
                return;
            }
            const taskId = args[1];
            markDone(taskId);
            break;
        default:
            showHelp();
    }
}

main();
```
