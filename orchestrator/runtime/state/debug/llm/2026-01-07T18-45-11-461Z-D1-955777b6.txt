```javascript
const fs = require('fs');
const path = require('path');

const tasksPath = path.join(__dirname, 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (err) {
        if (err.code === 'ENOENT') {
            return [];
        }
        throw err;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: tasks.length ? tasks[tasks.length - 1].id + 1 : 1,
        text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly.');
        return;
    }
    tasks.forEach((task, index) => {
        const status = task.done ? '[x]' : '[ ]';
        console.log(`${index + 1}. ${status} ${task.text}`);
    });
}

function doneTask(id) {
    const tasks = loadTasks();
    const taskId = parseInt(id);
    if (isNaN(taskId)) {
        console.error('Neplatné ID úkolu.');
        return;
    }
    const taskIndex = tasks.findIndex(task => task.id === taskId);
    if (taskIndex === -1) {
        console.error('Úkol s tímto ID neexistuje.');
        return;
    }
    tasks[taskIndex].done = true;
    saveTasks(tasks);
    console.log('Změněno na hotovo.');
    listTasks();
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.error('Zadejte příkaz: add <text>, list, done <id>');
        return;
    }

    const command = args[0];
    switch(command) {
        case 'add':
            if (args.length < 2) {
                console.error('Příkaz add vyžaduje text.');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;
        case 'list':
            listTasks();
            break;
        case 'done':
            if (args.length < 2) {
                console.error('Příkaz done vyžaduje ID.');
                return;
            }
            const taskID = args[1];
            doneTask(taskID);
            break;
        default:
            console.error('Neznámý příkaz. Možné příkazy: add, list, done');
    }
}

main();
```

  - Řešení: Při přidávání úkolu zkontrolujte počet úkolů. Pokud jsou nějaké, id = poslední id +1, jinak 1.

- Validace vstupu: Pro 'done <id>', zkontrolujte, že id existuje a je číslo.
