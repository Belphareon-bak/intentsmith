```javascript
// /tmp/workflow-auto/todo.js

const fs = require('fs');
const path = require('path');
const { program } = require('commander');

const tasksPath = path.join(__dirname, 'tasks.json');

program
  .version('1.0.0')
  .description('TODO CLI Aplikace');

program
  .command('add <text>')
  .description('Přidání nového úkolu')
  .action((text) => {
    addTask(text);
  });

program
  .command('list')
  .description('Výpis všech úkolů')
  .action(() => {
    listTasks();
  });

program
  .command('done <id>')
  .description('Označení úkolu jako hotové')
  .action((id) => {
    markTaskDone(id);
  });

program.parse(process.argv);

function addTask(text) {
  try {
    let tasks = [];
    if (fs.existsSync(tasksPath)) {
      tasks = JSON.parse(fs.readFileSync(tasksPath, 'utf8'));
    }

    const newTask = {
      id: tasks.length + 1,
      text: text,
      done: false,
      createdAt: new Date().toISOString()
    };

    tasks.push(newTask);
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
    console.log(`Úkol "${text}" byl přidán.`);
  } catch (error) {
    console.error('Chyba při přidávání úkolu:', error.message);
  }
}

function listTasks() {
  try {
    if (!fs.existsSync(tasksPath)) {
      console.log('Žádné úkoly.');
      return;
    }

    const tasks = JSON.parse(fs.readFileSync(tasksPath, 'utf8'));
    if (tasks.length === 0) {
      console.log('Žádné úkoly.');
      return;
    }

    tasks.forEach((task, index) => {
      const status = task.done ? '[x]' : '[ ]';
      console.log(`${index + 1}. ${status} ${task.text}`);
    });
  } catch (error) {
    console.error('Chyba při výpisu úkolů:', error.message);
  }
}

function markTaskDone(id) {
  try {
    if (!fs.existsSync(tasksPath)) {
      console.log('Žádné úkoly.');
      return;
    }

    const tasks = JSON.parse(fs.readFileSync(tasksPath, 'utf8'));
    const taskIndex = tasks.findIndex(task => task.id === parseInt(id));

    if (taskIndex === -1) {
      console.log('Úkol nebyl nalezen.');
      return;
    }

    tasks[taskIndex].done = true;
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
    console.log(`Úkol s ID ${id} byl označen jako hotový.`);
  } catch (error) {
    console.error('Chyba při označování úkolu jako hotové:', error.message);
  }
}
```
