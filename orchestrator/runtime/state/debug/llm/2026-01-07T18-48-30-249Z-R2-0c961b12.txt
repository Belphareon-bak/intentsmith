```javascript
// todo.js

const fs = require('fs');
const path = require('path');

const TASKS_FILE = path.join(__dirname, 'tasks.json');

function loadTasks() {
  try {
    const fileContent = fs.readFileSync(TASKS_FILE, 'utf8');
    return JSON.parse(fileContent);
  } catch (err) {
    return [];
  }
}

function saveTasks(tasks) {
  fs.writeFileSync(TASKS_FILE, JSON.stringify(tasks, null, 2));
}

function getNewId() {
  const tasks = loadTasks();
  return tasks.length === 0 ? 1 : Math.max(...tasks.map(task => task.id)) + 1;
}

function addTask(text) {
  const tasks = loadTasks();
  const newTask = {
    id: getNewId(),
    text,
    done: false,
    createdAt: Date.now()
  };
  tasks.push(newTask);
  saveTasks(tasks);
  console.log('Úkol byl přidán.');
}

function listTasks() {
  const tasks = loadTasks();
  if (tasks.length === 0) {
    console.log('Seznam úkolů je prázdný.');
    return;
  }
  console.log('\nSeznam úkolů:');
  tasks.forEach(task => {
    const status = task.done ? '[x]' : '[ ]';
    console.log(` ${task.id}. ${status} ${task.text}`);
  });
}

function markAsDone(id) {
  const tasks = loadTasks();
  const taskIndex = tasks.findIndex(task => task.id === id);
  if (taskIndex === -1) {
    console.log('Úkol s daným ID neexistuje.');
    return;
  }
  tasks[taskIndex].done = true;
  saveTasks(tasks);
  console.log('Úkol byl označen jako hotový.');
}

function main() {
  const args = process.argv.slice(2);
  if (args.length === 0) {
    console.log('Zadejte příkaz: add, list nebo done');
    return;
  }

  const command = args[0];
  switch (command.toLowerCase()) {
    case 'add':
      if (args.length < 2) {
        console.log('Prosím zadejte text úkolu.');
        return;
      }
      const taskText = args.slice(1).join(' ');
      addTask(taskText);
      break;

    case 'list':
      listTasks();
      break;

    case 'done':
      if (args.length < 2) {
        console.log('Prosím zadejte ID úkolu.');
        return;
      }
      const taskId = parseInt(args[1], 10);
      if (isNaN(taskId)) {
        console.log('ID musí být číslo.');
        return;
      }
      markAsDone(taskId);
      break;

    default:
      console.log('Neznámý příkaz. Dostupné příkazy: add, list, done');
  }
}

main();
```

### Popis Funkcí

- **loadTasks()**: Načte úkoly z `tasks.json` nebo vytvoří prázdné pole, pokud soubor neexistuje.
- **saveTasks(tasks)**: Uloží aktuální stav úkolů do `tasks.json`.
- **getNewId()**: Vrátí nové unikátní ID pro nový úkol.
- **addTask(text)**: Přidá nový úkol s daným textem, generuje nové ID a uloží ho.
- **listTasks()**: Vypíše seznam úkolů včetně jejich stavu (hotovo/nehotovo).
- **markAsDone(id)**: Označí úkol s daným ID jako hotový.
