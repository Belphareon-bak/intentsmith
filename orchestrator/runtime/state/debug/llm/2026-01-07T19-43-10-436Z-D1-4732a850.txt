```javascript
const fs = require('fs');
const path = require('path');

const tasksPath = path.join('/tmp', 'workflow-auto', 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (error) {
        if (error.code === 'ENOENT') {
            return [];
        }
        throw error;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2), 'utf8');
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: tasks.length + 1,
        text: text,
        done: false,
        createdAt: new Date().toISOString().split('T')[0]
    };
    tasks.push(newTask);
    saveTasks(tasks);
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly');
        return;
    }
    tasks.forEach(task => {
        const status = task.done ? '[x]' : '[ ]';
        console.log(`${task.id}. ${status} ${task.text} (${task.createdAt})`);
    });
}

function markDone(id) {
    const tasks = loadTasks();
    const index = tasks.findIndex(task => task.id === parseInt(id, 10));
    if (index === -1) {
        console.error('Úkol s daným ID neexistuje');
        return;
    }
    tasks[index].done = true;
    saveTasks(tasks);
}

function showHelp() {
    console.log(`
Použití: node todo.js [command]

Dostupné příkazy:
  add <text>   - Přidá nový úkol
  list        - Vypíše všechny úkoly
  done <id>   - Označí úkol jako hotový
  help        - Zobrazí nápovědu

Příklady:
  node todo.js add "Učit se Node.js"
  node todo.js list
  node todo.js done 1
`);
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        showHelp();
        return;
    }

    const command = args[0];
    switch (command.toLowerCase()) {
        case 'add':
            if (args.length < 2) {
                console.error('Chybí text úkolu');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;

        case 'list':
            listTasks();
            break;

        case 'done':
            if (args.length < 2) {
                console.error('Chybí ID úkolu');
                return;
            }
            const taskId = args[1];
            markDone(taskId);
            break;

        case 'help':
            showHelp();
            break;

        default:
            console.error('Neplatný příkaz. Zadejte "help" pro nápovědu.');
            break;
    }
}

// Ensure tasks.json exists with empty array if not present
try {
    fs.accessSync(tasksPath, fs.constants.F_OK);
} catch (error) {
    saveTasks([]);
}

main();
```
