### 3. Funkce pro ukládání úkolů
Implementujeme funkci `saveTasks`, která uloží úkoly do souboru.

```javascript
async function saveTasks(tasks: Task[]) {
    try {
        const data = JSON.stringify(tasks, null, 2);
        await fs.promises.writeFile(TASKS_FILE, data);
    } catch (error) {
        console.error('Chyba při ukládání úkolů:', error);
        throw error;
    }
}
```

### 4. Funkce pro načítání úkolů
Implementujeme funkci `loadTasks`, která načte úkoly z souboru.

```javascript
async function loadTasks(): Promise<Task[]> {
    try {
        const data = await fs.promises.readFile(TASKS_FILE, 'utf8');
        return JSON.parse(data) || [];
    } catch (error) {
        if ((error as NodeJS.ErrnoE).code === 'ENOENT') {
            // Soubor neexistuje, vytvoříme prázdný pole
            return [];
        }
        console.error('Chyba při načítání úkolů:', error);
        throw error;
    }
}
```

### 5. Implementace příkazů

#### Příkaz `add <text>`
Přidá nový úkol do seznamu.

```javascript
async function addTask(text: string) {
    const tasks = await loadTasks();
    const newTask: Task = {
        id: tasks.length + 1,
        text,
        done: false,
        createdAt: new Date().toISOString(),
    };
    tasks.push(newTask);
    await saveTasks(tasks);
    console.log('Úkol byl přidán.');
}
```

#### Příkaz `list`
Vypíše všechny úkoly s formátovaným výstupem.

```javascript
async function listTasks() {
    const tasks = await loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly nenalezeno.');
        return;
    }
    console.log('\nSeznam úkolů:');
    tasks.forEach((task, index) => {
        const doneSymbol = task.done ? '[x]' : '[ ]';
        console.log(`[${index + 1}] ${doneSymbol} ${task.text}`);
    });
}
```

#### Příkaz `done <id>`
Označí úkol jako hotový.

```javascript
async function markTaskAsDone(id: number) {
    const tasks = await loadTasks();
    const taskIndex = tasks.findIndex(task => task.id === id);
    if (taskIndex === -1) {
        console.error('Úkol s daným ID nebyl nalezen.');
        return;
    }
    tasks[taskIndex].done = true;
    await saveTasks(tasks);
    console.log('Úkol byl označen jako hotový.');
}
```

### 6. Parsování argumentů CLI
Zpracujeme příkazové parametry.

```javascript
async function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.error('Zadejte platný příkaz: add, list nebo done.');
        return;
    }

    const command = args[0];
    switch (command.toLowerCase()) {
        case 'add':
            if (args.length < 2) {
                console.error('Text úkolu je vyžadován.');
                return;
            }
            const text = args.slice(1).join(' ');
            await addTask(text);
            break;

        case 'list':
            await listTasks();
            break;

        case 'done':
            if (args.length < 2) {
                console.error('ID úkolu je vyžadováno.');
                return;
            }
            const taskId = parseInt(args[1], 10);
            if (isNaN(taskId)) {
                console.error('Neplatné ID úkolu.');
                return;
            }
            await markTaskAsDone(taskId);
            break;

        default:
            console.error('Neznámý příkaz. Použijte add, list nebo done.');
    }
}

main().catch(error => {
    console.error('Chyba:', error);
});
```
