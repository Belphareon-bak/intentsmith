Odešlu ti kód pro `todo.js`, který splňuje požadavky.

```javascript
const fs = require('fs');
const path = require('path');

const tasksPath = path.join('/tmp', 'workflow-auto', 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (error) {
        if (error.code === 'ENOENT') {
            return [];
        }
        throw error;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: Date.now().toString(),
        text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
    console.log('Task added successfully!');
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('No tasks found.');
        return;
    }
    console.log('\nList of tasks:');
    tasks.forEach(task => {
        const checkbox = task.done ? '[x]' : '[ ]';
        console.log(`ID: ${task.id}`);
        console.log(`Text: ${task.text}`);
        console.log(`Status: ${checkbox}`);
        console.log(`Created At: ${new Date(task.createdAt).toLocaleString()}\n`);
    });
}

function markTaskAsDone(id) {
    const tasks = loadTasks();
    const taskIndex = tasks.findIndex(task => task.id === id);
    if (taskIndex === -1) {
        console.log('Task not found.');
        return;
    }
    tasks[taskIndex].done = true;
    saveTasks(tasks);
    console.log('Task marked as done!');
}

function showHelp() {
    console.log(`
Usage: todo <command>
Available commands:
- add <text> - Add a new task
- list - List all tasks
- done <id> - Mark task as done
- help, ? - Show this help message
`);
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.log('No command provided. Try "help" for usage.');
        return;
    }

    const command = args[0];
    switch (command.toLowerCase()) {
        case 'add':
            if (args.length < 2) {
                console.log('Missing task text. Usage: todo add <text>');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;
        case 'list':
            listTasks();
            break;
        case 'done':
            if (args.length < 2) {
                console.log('Missing task ID. Usage: todo done <id>');
                return;
            }
            const taskId = args[1];
            markTaskAsDone(taskId);
            break;
        case 'help':
        case '?':
            showHelp();
            break;
        default:
            console.log('Invalid command. Try "help" for usage.');
    }
}

main();
```

Tento kód vytvoří jednoduchou CLI aplikaci pro správu úkolů s následujícими funkcemi:

1. Přidání úkolu (`add <text>`)
2. Zobrazení seznamu úkolů (`list`)
3. Označení úkolu jako hotového (`done <id>`)
4. Nápověda (`help` nebo `?`)

Úkoly jsou ukládány do souboru `/tmp/workflow-auto/tasks.json` ve formátu JSON, každý úkol obsahuje:
- id (generované číslo)
- text
- done (boolean)
- createdAt (datum vytvoření)

Pro spuštění aplikace z terminálu použijte:

```bash
node todo.js <command>
```

Příklady použití:
- Přidání úkolu: `node todo.js add "Buy groceries"`
- Zobrazení úkolů: `node todo.js list`
- Označení úkolu jako hotového: `node todo.js done <task_id>`
