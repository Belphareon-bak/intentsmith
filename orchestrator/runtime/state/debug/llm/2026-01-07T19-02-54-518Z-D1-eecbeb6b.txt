#### Krok 2: Vytvořte soubor `app.py`

```python
from flask import Flask, render_template, request, send_file
import os
import pandas as pd
from werkzeug.utils import secure_filename

app = Flask(__name__)
UPLOAD_FOLDER = 'uploads'
ALLOWED_EXTENSIONS = {'txt', 'csv'}

if not os.path.exists(UPLOAD_FOLDER):
    os.makedirs(UPLOAD_FOLDER)

def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/upload', methods=['POST'])
def upload_file():
    if request.method == 'POST':
        file = request.files['file']
        if file and allowed_file(file.filename):
            filename = secure_filename(file.filename)
            filepath = os.path.join(UPLOAD_FOLDER, filename)
            file.save(filepath)
            
            # Read the file
            df = pd.read_csv(filepath, header=None)
            
            return render_template('display.html', data=df.to_html(index=False), filename=filename)
    return 'File not uploaded'

@app.route('/analyze/<filename>')
def analyze(filename):
    filepath = os.path.join(UPLOAD_FOLDER, filename)
    with open(filepath, 'r') as f:
        content = f.read()
    
    num_chars = len(content)
    num_words = len(content.split())
    num_lines = content.count('\n')
    
    return render_template('analyze.html', 
                         chars=num_chars,
                         words=num_words,
                         lines=num_lines)

@app.route('/search/<filename>')
def search(filename):
    query = request.args.get('query', '')
    filepath = os.path.join(UPLOAD_FOLDER, filename)
    with open(filepath, 'r') as f:
        content = f.read()
    
    matches = [line.strip() for line in content.split('\n') if query in line]
    
    return render_template('search.html', 
                         results=matches,
                         query=query)

@app.route('/export/<filename>')
def export(filename):
    filepath = os.path.join(UPLOAD_FOLDER, filename)
    df = pd.read_csv(filepath, header=None)
    excel_path = os.path.join('exports', f'{os.path.splitext(filename)[0]}.xlsx')
    
    if not os.path.exists('exports'):
        os.makedirs('exports')
    
    df.to_excel(excel_path, index=False, header=False)
    
    return send_file(excel_path,
                     as_attachment=True,
                     download_name=f'{os.path.splitext(filename)[0]}.xlsx')

if __name__ == '__main__':
    app.run(debug=True)
```
