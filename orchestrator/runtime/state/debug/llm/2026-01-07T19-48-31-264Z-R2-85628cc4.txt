```javascript
const fs = require('fs');
const path = require('path');

const tasksPath = path.join(__dirname, 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (err) {
        if (err.code === 'ENOENT') {
            // Soubor neexistuje, vytvoříme prázdné pole
            saveTasks([]);
            return [];
        }
        throw err;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: Date.now().toString(),
        text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
    console.log('Úkol byl přidán.');
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly nenalezeny.');
        return;
    }
    console.log('\nSeznam úkolů:');
    tasks.forEach((task, index) => {
        const status = task.done ? '[x]' : '[ ]';
        const date = new Date(task.createdAt).toLocaleDateString();
        console.log(`${index + 1}. ${status} ${task.text} (Vytvořeno: ${date})`);
    });
}

function markTaskAsDone(id) {
    const tasks = loadTasks();
    const foundIndex = tasks.findIndex(task => task.id === id);
    if (foundIndex === -1) {
        console.log('Úkol s daným ID nebyl nalezen.');
        return;
    }
    tasks[foundIndex].done = true;
    saveTasks(tasks);
    console.log('Úkol byl označen jako hotový.');
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.log('Zadejte příkaz: add <text>, list, nebo done <id>');
        return;
    }

    const command = args[0];
    switch (command) {
        case 'add':
            if (args.length < 2) {
                console.log('Prosím zadejte text úkolu.');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;
        case 'list':
            listTasks();
            break;
        case 'done':
            if (args.length < 2) {
                console.log('Prosím zadejte ID úkolu.');
                return;
            }
            const taskId = args[1];
            markTaskAsDone(taskId);
            break;
        default:
            console.log('Neznámý příkaz. Možné příkazy: add, list, done');
    }
}

main();
```

// /tmp/workflow-auto/todo.js
const fs = require('fs');
const path = require('path');

const tasksPath = path.join(__dirname, 'tasks.json');

function loadTasks() {
    try {
        const fileContent = fs.readFileSync(tasksPath, 'utf8');
        return JSON.parse(fileContent);
    } catch (err) {
        if (err.code === 'ENOENT') {
            // Soubor neexistuje, vytvoříme prázdné pole
            saveTasks([]);
            return [];
        }
        throw err;
    }
}

function saveTasks(tasks) {
    fs.writeFileSync(tasksPath, JSON.stringify(tasks, null, 2));
}

function addTask(text) {
    const tasks = loadTasks();
    const newTask = {
        id: Date.now().toString(),
        text,
        done: false,
        createdAt: new Date().toISOString()
    };
    tasks.push(newTask);
    saveTasks(tasks);
    console.log('Úkol byl přidán.');
}

function listTasks() {
    const tasks = loadTasks();
    if (tasks.length === 0) {
        console.log('Žádné úkoly nenalezeny.');
        return;
    }
    console.log('\nSeznam úkolů:');
    tasks.forEach((task, index) => {
        const status = task.done ? '[x]' : '[ ]';
        const date = new Date(task.createdAt).toLocaleDateString();
        console.log(`${index + 1}. ${status} ${task.text} (Vytvořeno: ${date})`);
    });
}

function markTaskAsDone(id) {
    const tasks = loadTasks();
    const foundIndex = tasks.findIndex(task => task.id === id);
    if (foundIndex === -1) {
        console.log('Úkol s daným ID nebyl nalezen.');
        return;
    }
    tasks[foundIndex].done = true;
    saveTasks(tasks);
    console.log('Úkol byl označen jako hotový.');
}

function main() {
    const args = process.argv.slice(2);
    if (args.length === 0) {
        console.log('Zadejte příkaz: add <text>, list, nebo done <id>');
        return;
    }

    const command = args[0];
    switch (command) {
        case 'add':
            if (args.length < 2) {
                console.log('Prosím zadejte text úkolu.');
                return;
            }
            const taskText = args.slice(1).join(' ');
            addTask(taskText);
            break;
        case 'list':
            listTasks();
            break;
        case 'done':
            if (args.length < 2) {
                console.log('Prosím zadejte ID úkolu.');
                return;
            }
            const taskId = args[1];
            markTaskAsDone(taskId);
            break;
        default:
            console.log('Neznámý příkaz. Možné příkazy: add, list, done');
    }
}

main();
```
