```javascript
const Database = require('better-sqlite3');
const db = new Database('/tmp/dual-test/notes.db', { verbose: console.log });

// Vytvoření tabulky pro poznámky, pokud neexistuje
db.exec(`
CREATE TABLE IF NOT EXISTS notes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    content TEXT NOT NULL,
    tag TEXT
);
`);

module.exports = {
    addNote(content, tag) {
        const stmt = db.prepare('INSERT INTO notes (content, tag) VALUES (?, ?)');
        return stmt.run(content, tag);
    },
    listNotes() {
        return db.prepare('SELECT * FROM notes').all();
    },
    searchNotes(query) {
        return db.prepare('SELECT * FROM notes WHERE content LIKE ? OR tag LIKE ?')
            .all(`%${query}%`, `%${query}%`);
    },
    deleteNote(id) {
        const stmt = db.prepare('DELETE FROM notes WHERE id = ?');
        return stmt.run(id);
    },
    exportNotes() {
        return db.prepare('SELECT * FROM notes').all();
    }
};
```

### Soubor `notes.js` (hlavní logika aplikace)

```javascript
const db = require('./db.js');
const args = process.argv.slice(2);

function help() {
    console.log(`
Usage: node notes.js <command> [args]

Commands:
  add <content> <tag>     Add a new note with a tag
  list                    List all notes
  search <query>          Search notes by content or tag
  delete <id>             Delete a note by ID
  export                  Export all notes to JSON
`);
}

function addNote(content, tag) {
    if (!content || !tag) {
        console.error('Content and tag are required.');
        return;
    }
    db.addNote(content, tag);
    console.log('Note added successfully.');
}

function listNotes() {
    const notes = db.listNotes();
    if (notes.length === 0) {
        console.log('No notes found.');
        return;
    }
    notes.forEach(note => {
        console.log(`ID: ${note.id}, Content: ${note.content}, Tag: ${note.tag}`);
    });
}

function searchNotes(query) {
    if (!query) {
        console.error('Search query is required.');
        return;
    }
    const results = db.searchNotes(query);
    if (results.length === 0) {
        console.log('No notes found matching the query.');
        return;
    }
    results.forEach(note => {
        console.log(`ID: ${note.id}, Content: ${note.content}, Tag: ${note.tag}`);
    });
}

function deleteNote(id) {
    if (!id) {
        console.error('Note ID is required.');
        return;
    }
    const info = db.deleteNote(id);
    if (info.changes === 0) {
        console.log('No note found with the given ID.');
        return;
    }
    console.log('Note deleted successfully.');
}

function exportNotes() {
    const notes = db.exportNotes();
    console.log(JSON.stringify(notes, null, 2));
}

switch (args[0]) {
    case 'add':
        addNote(args[1], args[2]);
        break;
    case 'list':
        listNotes();
        break;
    case 'search':
        searchNotes(args[1]);
        break;
    case 'delete':
        deleteNote(args[1]);
        break;
    case 'export':
        exportNotes();
        break;
    default:
        help();
}
```

```javascript
const db = require('./db.js');
const args = process.argv.slice(2);

function help() {
    console.log(`
Usage: node notes.js <command> [args]

Commands:
  add "<content>" "<tag>"   Add a new note with content and tag
  list                      List all notes in a formatted table
  search "<query>"          Search notes by content or tag

function addNote(content, tag) {

        console.log('Note added successfully.');
    } catch (error) {

    }

function listNotes() {
    try {
        const notes = db.listNotes();
        if (notes.length === 0) {
            console.log('No notes found.');

        if (!query) {

        }
```

- **Uložení databáze:** Databázový soubor je uložen v adresáři projektu `./notes.db`, což je bezpečnější než `/tmp/`.

## Citelnost

- **Formátované výstupy:** Funkce `list` a `search` zobrazují poznámky ve formátu tabulky.
- **Jasné chybové hlásky:** Všechny chyby jsou zobrazeny s podrobným popisem, aby uživatelé věděli, co se pokazilo.

## Výstup

Problémy v současném kódu:

1. **Chyby:**
   - **Nedodržení požadovaných argumentů:** Při volání funkcí jako `add`, `search`, `delete` nebo `export` není kontrola, zda jsou všechny požadované argumenty poskytnuty. To může vést k chybám runtime.
   - **Nesprávný formát ID při mazání:** Argument pro mazání poznámky je řetězec, což může způsobit problémy při porovnání s číselným ID v databázi.
   - **Nedostatečné logování chyb:** V současném kódu nejsou správně zpracovány a zobrazeny chyby, což může uživateli způsobit nesrozumitelné chybové hlásky.

2. **Bezpečnost:**
   - **SQL Injectio:** Používání interpolace řetězců v SQL dotazech bez ochrany může vést k SQL injectio, což je značný bezpečnostní riziko.
   - **Nesprávné ukládání cest k souborům:** Uložení databázového souboru v adresáři `/tmp/` může být nebezpečné, protože tento adresář je často automaticky vyprazdňován.

3. **Citelnost:**
   - **Nedostatečný výstup pro uživatele:** Výstupy funkcí jsou minimalizované a mohou být nejasné pro někoho, kdo je poprvé používá.
   - **Absence informací o chybách:** Uživatelé nemají dostatečných informací o tom, co se pokazilo a jak to napravit.

**Návrhy vylepšení:**

1. **Chyby:**
   - **Kontrola argumentů:** Před spuštěním libové funkce zkontrolujte počet a formát argumentů. Například, pro `add` musí být poskytnuty obsah a tag.
   - **Převod ID na číslo:** Argument pro mazání poznámky převeďte na číslo, aby se zajistilo správné porovnání s databází.
   - **Vylepšené logování chyb:** Zobrazujte uživatelům jasně formulované chybové hlásky, které vysvětlují, co se pokazilo a jak to napravit.
